graph = [
    [0,1,4],
    [0,7,8],
    [1,2,8],
    [1,7,11],
    [2,3,7],
    [2,8,2],
    [2,5,4],
    [3,4,9],
    [3,5,14],
    [4,5,10],
    [5,6,2],
    [6,7,1],
    [6,8,6],
    [7,8,7]
]



graph.sort(key = lambda x : x[2])

parent = {}

for i in range(9):
    parent[i] = i


def find(x):

    while parent[x] != x:

        x = parent[x]

    return x


print("Minimum Spanning Tree:")

total = 0

for u, v, weight in graph:

    parent_u = find(u)
    parent_v = find(v)

    if parent_u != parent_v:

        print(u, "-", v, "=", weight)

        total = total + weight

        parent[parent_v] = parent_u


print("Total Cost =", total)
