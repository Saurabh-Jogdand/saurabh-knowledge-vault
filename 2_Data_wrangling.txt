import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
df = pd.read_csv('academic_data.csv')
df.head()
df.shape
df.dtypes
df
# Replace inconsistent missing values
df.replace(['NA', 'N/A', 'na', 'N.A'], np.nan, inplace=True)

# Convert numeric columns
num_cols = ['SPOS', 'DSBDA', 'WT', 'DA']
for col in num_cols:
    df[col] = pd.to_numeric(df[col], errors='coerce')

# Check missing values
print(df.isnull().sum())
# Numeric → Median
for col in num_cols:
    df[col].fillna(df[col].median(), inplace=True)

# Categorical → Mode
df['Gender'].fillna(df['Gender'].mode()[0], inplace=True)
# Fix invalid values using clipping
for col in num_cols:
    df[col] = df[col].clip(0, 100)
Q1 = df['SPOS'].quantile(0.25)
Q3 = df['SPOS'].quantile(0.75)
IQR = Q3 - Q1

lower = Q1 - 1.5 * IQR
upper = Q3 + 1.5 * IQR

outliers = df[(df['SPOS'] < lower) | (df['SPOS'] > upper)]
print(outliers)
df['SPOS'] = np.where(df['SPOS'] > upper, upper,
                     np.where(df['SPOS'] < lower, lower, df['SPOS']))
# Reason: WT column has skewed distribution due to extreme values
df['log_WT'] = np.log(df['WT'])
# Reason: DSBDA values are large; sqrt reduces scale for better interpretation
df['sqrt_DSBDA'] = np.sqrt(df['DSBDA'])
# Reason: SPOS shows non-linear variation; power transform helps linearize it
df['power_SPOS'] = df['SPOS'] ** 1.5
df['SPOS'].hist()
plt.title("SPOS Distribution")
plt.show()
print(df.head())
print(df.describe())
